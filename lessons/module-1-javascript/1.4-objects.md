## 📖 What is an Object?

An object is a collection of **key-value pairs**. Think of it like a real-world object with properties.

```javascript
// A person has properties
const person = {
  name: "John",           // key: value
  age: 25,
  isStudent: true,
  hobbies: ["reading", "gaming"],
  address: {
    city: "New York",
    country: "USA"
  }
};
```

## 🔧 Creating Objects

### Object Literal (Most Common)

```javascript
const user = {
  name: "John",
  age: 25
};
```

### Object Constructor

```javascript
const user = new Object();
user.name = "John";
user.age = 25;
```

### Object.create()

```javascript
const personPrototype = {
  greet() {
    return `Hello, I'm ${this.name}`;
  }
};

const john = Object.create(personPrototype);
john.name = "John";
console.log(john.greet());  // "Hello, I'm John"
```

### Factory Function

```javascript
function createUser(name, age) {
  return {
    name,
    age,
    greet() {
      return `Hello, I'm ${this.name}`;
    }
  };
}

const user1 = createUser("John", 25);
const user2 = createUser("Jane", 30);
```

## 📥 Accessing Properties

### Dot Notation

```javascript
const user = { name: "John", age: 25 };

console.log(user.name);  // "John"
console.log(user.age);   // 25

// Set property
user.name = "Jane";
user.email = "jane@example.com";  // Add new property
```

### Bracket Notation

```javascript
const user = { name: "John", age: 25 };

console.log(user["name"]);  // "John"
console.log(user["age"]);   // 25

// Useful when key is dynamic or has special characters
const key = "name";
console.log(user[key]);  // "John"

// Keys with spaces or special characters
const obj = {
  "first name": "John",
  "my-property": 123
};
console.log(obj["first name"]);  // "John"
```

### Optional Chaining (?.)

```javascript
const user = {
  name: "John",
  address: {
    city: "New York"
  }
};

// Without optional chaining - crashes if address is undefined
// console.log(user.address.city);  // Works
// console.log(user.contact.phone); // Error!

// With optional chaining - returns undefined instead of crashing
console.log(user.address?.city);    // "New York"
console.log(user.contact?.phone);   // undefined (no error)

// Chain multiple levels
console.log(user.address?.street?.name);  // undefined

// With methods
const result = user.getName?.();  // undefined if getName doesn't exist
```

### Nullish Coalescing with Objects

```javascript
const user = {
  name: "John",
  age: 0,
  nickname: ""
};

// || returns first truthy value (0 and "" are falsy!)
console.log(user.age || 25);       // 25 (wrong if 0 is valid)
console.log(user.nickname || "N/A"); // "N/A" (wrong if "" is valid)

// ?? returns first non-null/undefined value
console.log(user.age ?? 25);       // 0 (correct)
console.log(user.nickname ?? "N/A"); // "" (correct)
console.log(user.missing ?? "default"); // "default"
```

## 🔄 Iterating Over Objects

### Object.keys() - Get All Keys

```javascript
const user = { name: "John", age: 25, city: "NYC" };

const keys = Object.keys(user);
console.log(keys);  // ["name", "age", "city"]

// Loop through keys
Object.keys(user).forEach(key => {
  console.log(`${key}: ${user[key]}`);
});
```

### Object.values() - Get All Values

```javascript
const user = { name: "John", age: 25, city: "NYC" };

const values = Object.values(user);
console.log(values);  // ["John", 25, "NYC"]

// Sum all numeric values
const scores = { math: 90, english: 85, science: 92 };
const total = Object.values(scores).reduce((sum, score) => sum + score, 0);
console.log(total);  // 267
```

### Object.entries() - Get Key-Value Pairs

```javascript
const user = { name: "John", age: 25, city: "NYC" };

const entries = Object.entries(user);
console.log(entries);
// [["name", "John"], ["age", 25], ["city", "NYC"]]

// Loop with destructuring
Object.entries(user).forEach(([key, value]) => {
  console.log(`${key}: ${value}`);
});

// Convert to Map
const userMap = new Map(Object.entries(user));
```

### for...in Loop

```javascript
const user = { name: "John", age: 25 };

for (const key in user) {
  console.log(`${key}: ${user[key]}`);
}

// Warning: for...in also iterates inherited properties
// Use hasOwnProperty to check
for (const key in user) {
  if (user.hasOwnProperty(key)) {
    console.log(`${key}: ${user[key]}`);
  }
}
```

## 🎁 Destructuring

### Basic Destructuring

```javascript
const user = { name: "John", age: 25, city: "NYC" };

// Old way
const name = user.name;
const age = user.age;

// Destructuring
const { name, age, city } = user;
console.log(name);  // "John"
console.log(age);   // 25
console.log(city);  // "NYC"
```

### Renaming Variables

```javascript
const user = { name: "John", age: 25 };

// Rename while destructuring
const { name: userName, age: userAge } = user;
console.log(userName);  // "John"
console.log(userAge);   // 25
```

### Default Values

```javascript
const user = { name: "John" };

const { name, age = 18, city = "Unknown" } = user;
console.log(name);  // "John"
console.log(age);   // 18 (default)
console.log(city);  // "Unknown" (default)
```

### Nested Destructuring

```javascript
const user = {
  name: "John",
  address: {
    city: "NYC",
    country: "USA"
  }
};

const { name, address: { city, country } } = user;
console.log(name);     // "John"
console.log(city);     // "NYC"
console.log(country);  // "USA"
```

### Rest in Destructuring

```javascript
const user = { name: "John", age: 25, city: "NYC", country: "USA" };

const { name, ...rest } = user;
console.log(name);  // "John"
console.log(rest);  // { age: 25, city: "NYC", country: "USA" }
```

### Function Parameter Destructuring

```javascript
// Instead of this
function greet(user) {
  return `Hello, ${user.name}! You are ${user.age} years old.`;
}

// Do this
function greet({ name, age }) {
  return `Hello, ${name}! You are ${age} years old.`;
}

// With defaults
function greet({ name = "Guest", age = 0 } = {}) {
  return `Hello, ${name}! You are ${age} years old.`;
}

greet({ name: "John", age: 25 });
greet({ name: "Jane" });  // age defaults to 0
greet();  // Both default
```

## 🔀 Spread Operator with Objects

### Copying Objects

```javascript
const original = { name: "John", age: 25 };

// Shallow copy
const copy = { ...original };
copy.name = "Jane";

console.log(original.name);  // "John" (unchanged)
console.log(copy.name);      // "Jane"
```

### Merging Objects

```javascript
const defaults = { theme: "light", language: "en" };
const userPrefs = { theme: "dark" };

const settings = { ...defaults, ...userPrefs };
console.log(settings);  // { theme: "dark", language: "en" }

// Later properties override earlier ones
```

### Adding/Overriding Properties

```javascript
const user = { name: "John", age: 25 };

// Add property
const userWithEmail = { ...user, email: "john@example.com" };

// Override property
const olderUser = { ...user, age: 26 };

// Both
const updatedUser = { ...user, age: 26, city: "NYC" };
```

## 🔧 Object Methods

### Object.assign()

```javascript
// Copy properties from source to target
const target = { a: 1 };
const source = { b: 2 };

Object.assign(target, source);
console.log(target);  // { a: 1, b: 2 }

// Multiple sources
const result = Object.assign({}, obj1, obj2, obj3);

// Note: Spread is usually preferred
const result = { ...obj1, ...obj2, ...obj3 };
```

### Object.freeze()

```javascript
const user = { name: "John", age: 25 };
Object.freeze(user);

user.name = "Jane";  // Silently fails (or throws in strict mode)
user.city = "NYC";   // Silently fails

console.log(user);  // { name: "John", age: 25 }

// Check if frozen
console.log(Object.isFrozen(user));  // true

// Note: Shallow freeze only!
const nested = { data: { value: 1 } };
Object.freeze(nested);
nested.data.value = 2;  // This WORKS! Nested object not frozen
```

### Object.seal()

```javascript
const user = { name: "John", age: 25 };
Object.seal(user);

user.name = "Jane";  // Works - can modify existing
user.city = "NYC";   // Fails - can't add new
delete user.age;     // Fails - can't delete

console.log(user);  // { name: "Jane", age: 25 }
```

### Object.fromEntries()

```javascript
// Convert entries back to object
const entries = [["name", "John"], ["age", 25]];
const obj = Object.fromEntries(entries);
console.log(obj);  // { name: "John", age: 25 }

// Useful with Map
const map = new Map([["a", 1], ["b", 2]]);
const objFromMap = Object.fromEntries(map);

// Transform object values
const prices = { apple: 1, banana: 2, orange: 3 };
const doubled = Object.fromEntries(
  Object.entries(prices).map(([key, value]) => [key, value * 2])
);
console.log(doubled);  // { apple: 2, banana: 4, orange: 6 }
```

## ⚠️ Reference vs Value

### Primitives are Copied by Value

```javascript
let a = 5;
let b = a;  // Copy the value

b = 10;
console.log(a);  // 5 (unchanged)
console.log(b);  // 10
```

### Objects are Copied by Reference

```javascript
const user1 = { name: "John" };
const user2 = user1;  // Copy the reference, not the object!

user2.name = "Jane";
console.log(user1.name);  // "Jane" (changed!)
console.log(user2.name);  // "Jane"

// Both variables point to the SAME object in memory
```

### Shallow vs Deep Copy

```javascript
const original = {
  name: "John",
  address: {
    city: "NYC"
  }
};

// Shallow copy - nested objects still shared
const shallow = { ...original };
shallow.name = "Jane";           // OK - doesn't affect original
shallow.address.city = "LA";     // BAD - affects original!

console.log(original.address.city);  // "LA" (changed!)

// Deep copy - completely independent
const deep = JSON.parse(JSON.stringify(original));
// Or use structuredClone (modern browsers)
const deep2 = structuredClone(original);

deep.address.city = "Chicago";
console.log(original.address.city);  // "LA" (unchanged)
```

### Comparing Objects

```javascript
const obj1 = { name: "John" };
const obj2 = { name: "John" };
const obj3 = obj1;

console.log(obj1 === obj2);  // false (different objects)
console.log(obj1 === obj3);  // true (same reference)

// Compare by content
const isEqual = JSON.stringify(obj1) === JSON.stringify(obj2);
console.log(isEqual);  // true

// Better: use a deep equality function or library
```

## 🏗️ Shorthand Syntax

### Property Shorthand

```javascript
const name = "John";
const age = 25;

// Old way
const user = { name: name, age: age };

// Shorthand - when variable name matches key
const user = { name, age };
console.log(user);  // { name: "John", age: 25 }
```

### Method Shorthand

```javascript
// Old way
const user = {
  name: "John",
  greet: function() {
    return `Hello, ${this.name}`;
  }
};

// Shorthand
const user = {
  name: "John",
  greet() {
    return `Hello, ${this.name}`;
  }
};
```

### Computed Property Names

```javascript
const key = "name";
const value = "John";

// Dynamic key
const user = {
  [key]: value,
  [`${key}Length`]: value.length
};

console.log(user);  // { name: "John", nameLength: 4 }

// Useful for creating objects dynamically
function createObject(key, value) {
  return { [key]: value };
}
```

## 🎯 Summary

- Objects store key-value pairs
- Access with dot notation (obj.key) or brackets (obj["key"])
- Use optional chaining (?.) to safely access nested properties
- Object.keys/values/entries for iteration
- Destructuring extracts properties into variables
- Spread operator (...) for copying and merging
- Objects are passed by reference, not value
- Use structuredClone() for deep copies
