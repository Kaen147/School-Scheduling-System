## 🔧 Environment Variables

### Backend (.env)

```env
# backend/.env
PORT=5000
MONGODB_URI=mongodb://localhost:27017/myapp
JWT_SECRET=your-super-secret-key
NODE_ENV=development
```

```javascript
// backend/server.js
require('dotenv').config();

const PORT = process.env.PORT || 5000;
const MONGODB_URI = process.env.MONGODB_URI;
```

### Frontend (.env)

```env
# frontend/.env
VITE_API_URL=http://localhost:5000
```

```javascript
// frontend/src/services/api.js
const api = axios.create({
  baseURL: import.meta.env.VITE_API_URL
});
```

### Why Environment Variables?

1. **Security** - Keep secrets out of code
2. **Flexibility** - Different values for dev/prod
3. **Configuration** - Easy to change without code changes

## 🌐 CORS (Cross-Origin Resource Sharing)

When frontend (port 5173) calls backend (port 5000), browsers block it by default. CORS allows it.

### Backend CORS Setup

```javascript
// backend/server.js
const cors = require('cors');

// Simple - allow all origins (development only!)
app.use(cors());

// Better - specify allowed origins
app.use(cors({
  origin: ['http://localhost:5173', 'http://localhost:3000'],
  credentials: true  // Allow cookies
}));

// Production - dynamic origin
app.use(cors({
  origin: function(origin, callback) {
    const allowedOrigins = [
      'http://localhost:5173',
      'https://myapp.vercel.app'
    ];
    
    // Allow requests with no origin (mobile apps, Postman)
    if (!origin) return callback(null, true);
    
    if (allowedOrigins.includes(origin)) {
      callback(null, true);
    } else {
      callback(new Error('Not allowed by CORS'));
    }
  },
  credentials: true
}));
```

### Common CORS Errors

```
Access to fetch at 'http://localhost:5000/api/users' from origin 
'http://localhost:5173' has been blocked by CORS policy
```

**Solutions:**
1. Add `cors()` middleware to backend
2. Check origin is in allowed list
3. Ensure backend is running
4. Check URL is correct (no typos)

## 📡 API Service Setup

```javascript
// frontend/src/services/api.js
import axios from 'axios';

const api = axios.create({
  baseURL: import.meta.env.VITE_API_URL || 'http://localhost:5000',
  timeout: 10000,
  headers: {
    'Content-Type': 'application/json'
  }
});

// Request interceptor - add auth token
api.interceptors.request.use(
  (config) => {
    const token = localStorage.getItem('token');
    if (token) {
      config.headers.Authorization = `Bearer ${token}`;
    }
    return config;
  },
  (error) => {
    return Promise.reject(error);
  }
);

// Response interceptor - handle errors globally
api.interceptors.response.use(
  (response) => response,
  (error) => {
    // Network error
    if (!error.response) {
      console.error('Network error - server may be down');
      return Promise.reject(new Error('Unable to connect to server'));
    }
    
    // Auth error
    if (error.response.status === 401) {
      localStorage.removeItem('token');
      window.location.href = '/login';
    }
    
    // Server error
    if (error.response.status >= 500) {
      console.error('Server error:', error.response.data);
    }
    
    return Promise.reject(error);
  }
);

export default api;
```

## 🔄 Consistent API Responses

### Backend Response Format

```javascript
// backend/utils/response.js
const sendSuccess = (res, data, message = 'Success', statusCode = 200) => {
  res.status(statusCode).json({
    success: true,
    message,
    data
  });
};

const sendError = (res, message = 'Error', statusCode = 500, errors = null) => {
  res.status(statusCode).json({
    success: false,
    message,
    errors
  });
};

module.exports = { sendSuccess, sendError };
```

```javascript
// backend/controllers/userController.js
const { sendSuccess, sendError } = require('../utils/response');

const getUsers = async (req, res) => {
  try {
    const users = await User.find();
    sendSuccess(res, users, 'Users fetched successfully');
  } catch (error) {
    sendError(res, 'Failed to fetch users', 500);
  }
};

const createUser = async (req, res) => {
  try {
    const user = await User.create(req.body);
    sendSuccess(res, user, 'User created successfully', 201);
  } catch (error) {
    if (error.code === 11000) {
      sendError(res, 'Email already exists', 400);
    } else {
      sendError(res, 'Failed to create user', 500);
    }
  }
};
```

### Frontend Handling

```javascript
// frontend/src/pages/Users.jsx
const fetchUsers = async () => {
  try {
    const { data } = await api.get('/api/users');
    
    if (data.success) {
      setUsers(data.data);
    }
  } catch (error) {
    // Error already handled by interceptor
    setError(error.response?.data?.message || 'Something went wrong');
  }
};
```

## 🔍 Debugging Connection Issues

### 1. Check Backend is Running

```bash
# Terminal 1 - Backend
cd backend
npm start
# Should see: "Server running on port 5000"
```

### 2. Test API Directly

```bash
# Using curl
curl http://localhost:5000/api/users

# Or open in browser
http://localhost:5000/api/users
```

### 3. Check Network Tab

In browser DevTools → Network tab:
- Look for red failed requests
- Check request URL
- Check response status and body

### 4. Console Logging

```javascript
// Backend - log incoming requests
app.use((req, res, next) => {
  console.log(`${req.method} ${req.path}`);
  next();
});

// Frontend - log API calls
api.interceptors.request.use((config) => {
  console.log('API Request:', config.method, config.url);
  return config;
});
```

### 5. Common Issues Checklist

| Issue | Solution |
|-------|----------|
| CORS error | Add cors middleware to backend |
| 404 Not Found | Check API URL and route |
| 401 Unauthorized | Check token is being sent |
| Network Error | Backend not running |
| Wrong data | Check request body format |
