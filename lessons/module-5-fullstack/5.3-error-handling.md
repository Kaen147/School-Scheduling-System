## 📖 Why Error Handling Matters

Without proper error handling:
- Users see cryptic messages or blank screens
- Developers can't debug issues
- Security vulnerabilities get exposed
- App crashes unexpectedly

With proper error handling:
- Users get helpful feedback
- Errors are logged for debugging
- App recovers gracefully
- Security is maintained

## 🔧 Backend Error Handling

### Error Types

```javascript
// Custom error classes
class AppError extends Error {
  constructor(message, statusCode) {
    super(message);
    this.statusCode = statusCode;
    this.isOperational = true;  // Expected error
  }
}

class ValidationError extends AppError {
  constructor(message) {
    super(message, 400);
  }
}

class NotFoundError extends AppError {
  constructor(resource = 'Resource') {
    super(`${resource} not found`, 404);
  }
}

class UnauthorizedError extends AppError {
  constructor(message = 'Not authorized') {
    super(message, 401);
  }
}

module.exports = { AppError, ValidationError, NotFoundError, UnauthorizedError };
```

### Using Custom Errors

```javascript
// controllers/todoController.js
const { NotFoundError, ValidationError } = require('../utils/errors');

const getTodo = async (req, res, next) => {
  try {
    const todo = await Todo.findById(req.params.id);
    
    if (!todo) {
      throw new NotFoundError('Todo');
    }
    
    res.json(todo);
  } catch (error) {
    next(error);  // Pass to error handler
  }
};

const createTodo = async (req, res, next) => {
  try {
    const { title } = req.body;
    
    if (!title || title.trim() === '') {
      throw new ValidationError('Title is required');
    }
    
    const todo = await Todo.create({ title, user: req.user.id });
    res.status(201).json(todo);
  } catch (error) {
    next(error);
  }
};
```

### Global Error Handler

```javascript
// middleware/errorHandler.js
const errorHandler = (err, req, res, next) => {
  // Log error for debugging
  console.error('Error:', {
    message: err.message,
    stack: err.stack,
    path: req.path,
    method: req.method
  });
  
  // Mongoose validation error
  if (err.name === 'ValidationError') {
    const messages = Object.values(err.errors).map(e => e.message);
    return res.status(400).json({
      success: false,
      message: 'Validation failed',
      errors: messages
    });
  }
  
  // Mongoose duplicate key error
  if (err.code === 11000) {
    const field = Object.keys(err.keyValue)[0];
    return res.status(400).json({
      success: false,
      message: `${field} already exists`
    });
  }
  
  // Mongoose cast error (invalid ObjectId)
  if (err.name === 'CastError') {
    return res.status(400).json({
      success: false,
      message: 'Invalid ID format'
    });
  }
  
  // JWT errors
  if (err.name === 'JsonWebTokenError') {
    return res.status(401).json({
      success: false,
      message: 'Invalid token'
    });
  }
  
  if (err.name === 'TokenExpiredError') {
    return res.status(401).json({
      success: false,
      message: 'Token expired'
    });
  }
  
  // Our custom AppError
  if (err.isOperational) {
    return res.status(err.statusCode).json({
      success: false,
      message: err.message
    });
  }
  
  // Unknown error - don't leak details in production
  res.status(500).json({
    success: false,
    message: process.env.NODE_ENV === 'production' 
      ? 'Something went wrong' 
      : err.message
  });
};

module.exports = errorHandler;
```

### Using Error Handler

```javascript
// server.js
const errorHandler = require('./middleware/errorHandler');

// Routes
app.use('/api/todos', todoRoutes);
app.use('/api/users', userRoutes);

// 404 handler
app.use((req, res) => {
  res.status(404).json({
    success: false,
    message: `Route ${req.path} not found`
  });
});

// Global error handler (must be last!)
app.use(errorHandler);
```

## 🎨 Frontend Error Handling

### API Error Handling

```javascript
// services/api.js
api.interceptors.response.use(
  (response) => response,
  (error) => {
    // No response - network error
    if (!error.response) {
      return Promise.reject({
        message: 'Unable to connect to server. Please check your internet connection.',
        isNetworkError: true
      });
    }
    
    const { status, data } = error.response;
    
    // Format error consistently
    const formattedError = {
      status,
      message: data.message || 'Something went wrong',
      errors: data.errors || []
    };
    
    // Handle specific status codes
    if (status === 401) {
      localStorage.removeItem('token');
      window.location.href = '/login';
    }
    
    return Promise.reject(formattedError);
  }
);
```

### Component Error Handling

```jsx
function TodoList() {
  const [todos, setTodos] = useState([]);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState(null);
  
  useEffect(() => {
    const fetchTodos = async () => {
      try {
        setError(null);
        const { data } = await api.get('/api/todos');
        setTodos(data);
      } catch (err) {
        setError(err.message);
      } finally {
        setLoading(false);
      }
    };
    
    fetchTodos();
  }, []);
  
  if (loading) return <Spinner />;
  
  if (error) {
    return (
      <div className="error-container">
        <p>{error}</p>
        <button onClick={() => window.location.reload()}>
          Try Again
        </button>
      </div>
    );
  }
  
  return <ul>{todos.map(t => <TodoItem key={t._id} todo={t} />)}</ul>;
}
```

### Form Error Handling

```jsx
function CreateTodo() {
  const [title, setTitle] = useState('');
  const [error, setError] = useState('');
  const [fieldErrors, setFieldErrors] = useState({});
  const [loading, setLoading] = useState(false);
  
  const handleSubmit = async (e) => {
    e.preventDefault();
    setError('');
    setFieldErrors({});
    
    // Client-side validation
    if (!title.trim()) {
      setFieldErrors({ title: 'Title is required' });
      return;
    }
    
    setLoading(true);
    
    try {
      await api.post('/api/todos', { title });
      setTitle('');
      // Success!
    } catch (err) {
      if (err.errors) {
        // Field-specific errors from server
        const errors = {};
        err.errors.forEach(e => {
          errors[e.field] = e.message;
        });
        setFieldErrors(errors);
      } else {
        // General error
        setError(err.message);
      }
    } finally {
      setLoading(false);
    }
  };
  
  return (
    <form onSubmit={handleSubmit}>
      {error && <div className="error">{error}</div>}
      
      <div className="form-group">
        <input
          value={title}
          onChange={(e) => setTitle(e.target.value)}
          className={fieldErrors.title ? 'input-error' : ''}
        />
        {fieldErrors.title && (
          <span className="field-error">{fieldErrors.title}</span>
        )}
      </div>
      
      <button disabled={loading}>
        {loading ? 'Creating...' : 'Create'}
      </button>
    </form>
  );
}
```

## 🛡️ Error Boundaries

Catch JavaScript errors in component tree:

```jsx
// components/ErrorBoundary.jsx
import { Component } from 'react';

class ErrorBoundary extends Component {
  constructor(props) {
    super(props);
    this.state = { hasError: false, error: null };
  }
  
  static getDerivedStateFromError(error) {
    return { hasError: true, error };
  }
  
  componentDidCatch(error, errorInfo) {
    // Log to error reporting service
    console.error('Error caught by boundary:', error, errorInfo);
  }
  
  render() {
    if (this.state.hasError) {
      return (
        <div className="error-boundary">
          <h2>Something went wrong</h2>
          <p>We're sorry, but something unexpected happened.</p>
          <button onClick={() => window.location.reload()}>
            Reload Page
          </button>
        </div>
      );
    }
    
    return this.props.children;
  }
}

export default ErrorBoundary;
```

## 📊 Error Display Components

### Toast Notifications

```jsx
// components/Toast.jsx
function Toast({ message, type = 'error', onClose }) {
  useEffect(() => {
    const timer = setTimeout(onClose, 5000);
    return () => clearTimeout(timer);
  }, [onClose]);
  
  return (
    <div className={`toast toast-${type}`}>
      <span>{message}</span>
      <button onClick={onClose}>×</button>
    </div>
  );
}

// Usage with context
function ToastProvider({ children }) {
  const [toasts, setToasts] = useState([]);
  
  const addToast = (message, type = 'error') => {
    const id = Date.now();
    setToasts(prev => [...prev, { id, message, type }]);
  };
  
  const removeToast = (id) => {
    setToasts(prev => prev.filter(t => t.id !== id));
  };
  
  return (
    <ToastContext.Provider value={{ addToast }}>
      {children}
      <div className="toast-container">
        {toasts.map(toast => (
          <Toast
            key={toast.id}
            message={toast.message}
            type={toast.type}
            onClose={() => removeToast(toast.id)}
          />
        ))}
      </div>
    </ToastContext.Provider>
  );
}
```

### Inline Error Messages

```jsx
function FormField({ label, name, error, ...props }) {
  return (
    <div className="form-field">
      <label htmlFor={name}>{label}</label>
      <input
        id={name}
        name={name}
        className={error ? 'input-error' : ''}
        aria-invalid={!!error}
        aria-describedby={error ? `${name}-error` : undefined}
        {...props}
      />
      {error && (
        <span id={`${name}-error`} className="error-message" role="alert">
          {error}
        </span>
      )}
    </div>
  );
}
```

## 🔍 Debugging Errors

### Backend Logging

```javascript
// utils/logger.js
const logger = {
  info: (message, data = {}) => {
    console.log(`[INFO] ${new Date().toISOString()} - ${message}`, data);
  },
  
  error: (message, error = {}) => {
    console.error(`[ERROR] ${new Date().toISOString()} - ${message}`, {
      message: error.message,
      stack: error.stack,
      ...error
    });
  },
  
  warn: (message, data = {}) => {
    console.warn(`[WARN] ${new Date().toISOString()} - ${message}`, data);
  }
};

module.exports = logger;
```

### Frontend Debugging

```jsx
// In development, log all errors
if (process.env.NODE_ENV === 'development') {
  api.interceptors.response.use(
    response => response,
    error => {
      console.group('API Error');
      console.log('URL:', error.config?.url);
      console.log('Method:', error.config?.method);
      console.log('Status:', error.response?.status);
      console.log('Data:', error.response?.data);
      console.groupEnd();
      return Promise.reject(error);
    }
  );
}
```
