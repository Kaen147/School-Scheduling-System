## 📖 Types of State

### 1. Local State
Data used by one component only.

```jsx
function Counter() {
  const [count, setCount] = useState(0);  // Local state
  return <button onClick={() => setCount(c => c + 1)}>{count}</button>;
}
```

### 2. Shared State
Data used by multiple components (lift state up).

```jsx
function Parent() {
  const [filter, setFilter] = useState('all');  // Shared between children
  
  return (
    <>
      <FilterButtons filter={filter} setFilter={setFilter} />
      <TodoList filter={filter} />
    </>
  );
}
```

### 3. Global State
Data needed throughout the app (auth, theme, etc.).

```jsx
// Use Context API
const AuthContext = createContext();

function App() {
  const [user, setUser] = useState(null);
  
  return (
    <AuthContext.Provider value={{ user, setUser }}>
      <Router />
    </AuthContext.Provider>
  );
}
```

### 4. Server State
Data from API that needs caching, syncing, etc.

```jsx
// Data fetched from backend
const [todos, setTodos] = useState([]);
const [loading, setLoading] = useState(true);
```

---

## 🎯 When to Use What

| State Type | Use When | Example |
|------------|----------|---------|
| useState | Single component needs it | Form input, toggle |
| Lift State | Parent + few children need it | Filter + list |
| Context | Many components need it | Auth, theme |
| URL State | Should be shareable/bookmarkable | Search, pagination |

## 🔧 Context API Deep Dive

### Creating Context

```jsx
// src/context/TodoContext.jsx
import { createContext, useContext, useState, useEffect } from 'react';
import api from '../services/api';

const TodoContext = createContext();

export function TodoProvider({ children }) {
  const [todos, setTodos] = useState([]);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState(null);
  
  // Fetch todos on mount
  useEffect(() => {
    fetchTodos();
  }, []);
  
  const fetchTodos = async () => {
    try {
      setLoading(true);
      const { data } = await api.get('/api/todos');
      setTodos(data);
      setError(null);
    } catch (err) {
      setError('Failed to fetch todos');
    } finally {
      setLoading(false);
    }
  };
  
  const addTodo = async (title) => {
    try {
      const { data } = await api.post('/api/todos', { title });
      setTodos(prev => [data, ...prev]);
    } catch (err) {
      setError('Failed to add todo');
      throw err;
    }
  };
  
  const updateTodo = async (id, updates) => {
    try {
      const { data } = await api.put(`/api/todos/${id}`, updates);
      setTodos(prev => prev.map(t => t._id === id ? data : t));
    } catch (err) {
      setError('Failed to update todo');
      throw err;
    }
  };
  
  const deleteTodo = async (id) => {
    try {
      await api.delete(`/api/todos/${id}`);
      setTodos(prev => prev.filter(t => t._id !== id));
    } catch (err) {
      setError('Failed to delete todo');
      throw err;
    }
  };
  
  const value = {
    todos,
    loading,
    error,
    addTodo,
    updateTodo,
    deleteTodo,
    refetch: fetchTodos
  };
  
  return (
    <TodoContext.Provider value={value}>
      {children}
    </TodoContext.Provider>
  );
}

// Custom hook for using context
export function useTodos() {
  const context = useContext(TodoContext);
  if (!context) {
    throw new Error('useTodos must be used within TodoProvider');
  }
  return context;
}
```

### Using Context

```jsx
// Any component in the tree
function TodoList() {
  const { todos, loading, deleteTodo } = useTodos();
  
  if (loading) return <Spinner />;
  
  return (
    <ul>
      {todos.map(todo => (
        <li key={todo._id}>
          {todo.title}
          <button onClick={() => deleteTodo(todo._id)}>Delete</button>
        </li>
      ))}
    </ul>
  );
}

function AddTodoForm() {
  const { addTodo } = useTodos();
  const [title, setTitle] = useState('');
  
  const handleSubmit = async (e) => {
    e.preventDefault();
    await addTodo(title);
    setTitle('');
  };
  
  return (
    <form onSubmit={handleSubmit}>
      <input value={title} onChange={e => setTitle(e.target.value)} />
      <button>Add</button>
    </form>
  );
}
```

## 🎣 Custom Hooks for State Logic

### useLocalStorage Hook

```jsx
function useLocalStorage(key, initialValue) {
  const [value, setValue] = useState(() => {
    const saved = localStorage.getItem(key);
    return saved ? JSON.parse(saved) : initialValue;
  });
  
  useEffect(() => {
    localStorage.setItem(key, JSON.stringify(value));
  }, [key, value]);
  
  return [value, setValue];
}

// Usage
function Settings() {
  const [theme, setTheme] = useLocalStorage('theme', 'light');
  
  return (
    <button onClick={() => setTheme(t => t === 'light' ? 'dark' : 'light')}>
      Current: {theme}
    </button>
  );
}
```

### useAsync Hook

```jsx
function useAsync(asyncFunction, immediate = true) {
  const [status, setStatus] = useState('idle');
  const [data, setData] = useState(null);
  const [error, setError] = useState(null);
  
  const execute = useCallback(async (...args) => {
    setStatus('pending');
    setError(null);
    
    try {
      const result = await asyncFunction(...args);
      setData(result);
      setStatus('success');
      return result;
    } catch (err) {
      setError(err);
      setStatus('error');
      throw err;
    }
  }, [asyncFunction]);
  
  useEffect(() => {
    if (immediate) {
      execute();
    }
  }, [execute, immediate]);
  
  return { execute, status, data, error, isLoading: status === 'pending' };
}

// Usage
function UserProfile({ userId }) {
  const { data: user, isLoading, error } = useAsync(
    () => api.get(`/users/${userId}`).then(r => r.data),
    true
  );
  
  if (isLoading) return <Spinner />;
  if (error) return <Error message={error.message} />;
  
  return <div>{user.name}</div>;
}
```

### useForm Hook

```jsx
function useForm(initialValues, validate) {
  const [values, setValues] = useState(initialValues);
  const [errors, setErrors] = useState({});
  const [touched, setTouched] = useState({});
  
  const handleChange = (e) => {
    const { name, value } = e.target;
    setValues(prev => ({ ...prev, [name]: value }));
  };
  
  const handleBlur = (e) => {
    const { name } = e.target;
    setTouched(prev => ({ ...prev, [name]: true }));
    
    if (validate) {
      const validationErrors = validate(values);
      setErrors(validationErrors);
    }
  };
  
  const reset = () => {
    setValues(initialValues);
    setErrors({});
    setTouched({});
  };
  
  const isValid = Object.keys(errors).length === 0;
  
  return {
    values,
    errors,
    touched,
    handleChange,
    handleBlur,
    reset,
    isValid,
    setValues
  };
}

// Usage
function LoginForm() {
  const { values, errors, touched, handleChange, handleBlur } = useForm(
    { email: '', password: '' },
    (values) => {
      const errors = {};
      if (!values.email) errors.email = 'Required';
      if (!values.password) errors.password = 'Required';
      return errors;
    }
  );
  
  return (
    <form>
      <input
        name="email"
        value={values.email}
        onChange={handleChange}
        onBlur={handleBlur}
      />
      {touched.email && errors.email && <span>{errors.email}</span>}
    </form>
  );
}
```

## 🔄 Combining Multiple Contexts

```jsx
// src/App.jsx
function App() {

    <AuthProvider>
```

### Avoiding Provider Hell

```jsx
// src/providers/AppProviders.jsx
function AppProviders({ children }) {
  return (
    <AuthProvider>
      <ThemeProvider>
        <TodoProvider>
          {children}
        </TodoProvider>
      </ThemeProvider>
    </AuthProvider>
  );
}

// src/App.jsx
function App() {
  return (
    <AppProviders>
      <Router />
    </AppProviders>
  );
}
```

## ⚠️ Common Mistakes

### 1. Putting Everything in Global State

```jsx
// ❌ Bad - form state doesn't need to be global
const GlobalContext = createContext();
function App() {
  const [formData, setFormData] = useState({});  // Don't do this!
}

// ✅ Good - keep form state local
function ContactForm() {
  const [formData, setFormData] = useState({});  // Local is fine
}
```

### 2. Not Memoizing Context Value

```jsx
// ❌ Bad - creates new object every render
function TodoProvider({ children }) {
  const [todos, setTodos] = useState([]);
  
  return (
    <TodoContext.Provider value={{ todos, setTodos }}>
      {children}
    </TodoContext.Provider>
  );
}

// ✅ Good - memoize the value
function TodoProvider({ children }) {
  const [todos, setTodos] = useState([]);
  
  const value = useMemo(() => ({ todos, setTodos }), [todos]);
  
  return (
    <TodoContext.Provider value={value}>
      {children}
    </TodoContext.Provider>
  );
}
```

### 3. Updating State Based on Old State

```jsx
// ❌ Bad - might use stale state
setCount(count + 1);
setCount(count + 1);  // Still uses old count!

// ✅ Good - use callback form
setCount(c => c + 1);
setCount(c => c + 1);  // Correctly increments twice
```

## 📊 State Management Decision Tree

```
Do you need this data?
│
├─► In one component only?
│   └─► useState
│
├─► In parent and children?
│   └─► Lift state to parent
│
├─► In many unrelated components?
│   └─► Context API
│
├─► Should it be in URL?
│   └─► useSearchParams / URL state
│
└─► Complex with many actions?
    └─► useReducer + Context
```

## 🎯 Summary

- **Local state** for single-component data
- **Lift state** when parent and children need it
- **Context** for truly global data (auth, theme)
- **Custom hooks** encapsulate reusable logic
- **Memoize** context values to prevent re-renders
- Keep state **as local as possible**
