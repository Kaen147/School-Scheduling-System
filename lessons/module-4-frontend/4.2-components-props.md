## 📖 Component Design Principles

### Single Responsibility

Each component should do one thing well.

```jsx
// ❌ Bad - does too much
function UserDashboard({ user }) {
  return (
    <div>
      <img src={user.avatar} />
      <h1>{user.name}</h1>
      <p>{user.bio}</p>
      <ul>
        {user.posts.map(post => (
          <li key={post.id}>
            <h3>{post.title}</h3>
            <p>{post.content}</p>
          </li>
        ))}
      </ul>
    </div>
  );
}

// ✅ Good - broken into focused components
function UserDashboard({ user }) {
  return (
    <div>
      <UserProfile user={user} />
      <PostList posts={user.posts} />
    </div>
  );
}

function UserProfile({ user }) {
  return (
    <div className="profile">
      <Avatar src={user.avatar} alt={user.name} />
      <h1>{user.name}</h1>
      <p>{user.bio}</p>
    </div>
  );
}

function PostList({ posts }) {
  return (
    <ul className="posts">
      {posts.map(post => (
        <PostItem key={post.id} post={post} />
      ))}
    </ul>
  );
}
```

## 📦 Props Deep Dive

### Passing Different Types

```jsx
function Example() {
  return (
    <Component
      // String
      title="Hello"
      
      // Number
      count={42}
      
      // Boolean
      isActive={true}
      isDisabled  // Same as isDisabled={true}
      
      // Array
      items={['a', 'b', 'c']}
      
      // Object
      user={{ name: 'John', age: 25 }}
      
      // Function
      onClick={() => console.log('clicked')}
      onSubmit={handleSubmit}
      
      // Component
      icon={<Icon />}
      
      // Render prop
      renderItem={(item) => <span>{item}</span>}
    />
  );
}
```

### Destructuring Props

```jsx
// In function parameter
function Button({ label, onClick, variant = 'primary', disabled = false }) {
  return (
    <button 
      className={`btn btn-${variant}`}
      onClick={onClick}
      disabled={disabled}
    >
      {label}
    </button>
  );
}

// In function body
function Button(props) {
  const { label, onClick, variant = 'primary', ...rest } = props;
  
  return (
    <button className={`btn btn-${variant}`} onClick={onClick} {...rest}>
      {label}
    </button>
  );
}
```

### Prop Spreading

```jsx
function Input(props) {
  const { label, error, ...inputProps } = props;
  
  return (
    <div className="form-group">
      <label>{label}</label>
      <input {...inputProps} />
      {error && <span className="error">{error}</span>}
    </div>
  );
}

// Usage - all standard input props work
<Input 
  label="Email"
  type="email"
  placeholder="Enter email"
  value={email}
  onChange={handleChange}
  required
/>
```

## 👶 Children Prop

### Basic Children

```jsx
function Card({ children }) {
  return <div className="card">{children}</div>;
}

// Usage
<Card>
  <h2>Title</h2>
  <p>Content goes here</p>
</Card>
```

### Children with Other Props

```jsx
function Modal({ title, children, onClose }) {
  return (
    <div className="modal-overlay">
      <div className="modal">
        <div className="modal-header">
          <h2>{title}</h2>
          <button onClick={onClose}>×</button>
        </div>
        <div className="modal-body">
          {children}
        </div>
      </div>
    </div>
  );
}

// Usage
<Modal title="Confirm Action" onClose={handleClose}>
  <p>Are you sure you want to proceed?</p>
  <button onClick={handleConfirm}>Yes</button>
  <button onClick={handleClose}>No</button>
</Modal>
```

### Manipulating Children

```jsx
import { Children, cloneElement } from 'react';

function ButtonGroup({ children }) {
  return (
    <div className="button-group">
      {Children.map(children, (child, index) => 
        cloneElement(child, {
          className: `${child.props.className || ''} grouped`,
          'data-index': index
        })
      )}
    </div>
  );
}
```

## 🎨 Component Patterns

### Container/Presentational

```jsx
// Container - handles logic
function UserListContainer() {
  const [users, setUsers] = useState([]);
  const [loading, setLoading] = useState(true);
  
  useEffect(() => {
    fetchUsers().then(data => {
      setUsers(data);
      setLoading(false);
    });
  }, []);
  
  if (loading) return <Spinner />;
  
  return <UserList users={users} />;
}

// Presentational - just renders UI
function UserList({ users }) {
  return (
    <ul>
      {users.map(user => (
        <UserItem key={user.id} user={user} />
      ))}
    </ul>
  );
}
```

### Compound Components

```jsx
function Tabs({ children, defaultTab }) {
  const [activeTab, setActiveTab] = useState(defaultTab);
  
  return (
    <TabsContext.Provider value={{ activeTab, setActiveTab }}>
      <div className="tabs">{children}</div>
    </TabsContext.Provider>
  );
}

Tabs.List = function TabList({ children }) {
  return <div className="tab-list">{children}</div>;
};

Tabs.Tab = function Tab({ id, children }) {
  const { activeTab, setActiveTab } = useContext(TabsContext);
  return (
    <button 
      className={activeTab === id ? 'active' : ''}
      onClick={() => setActiveTab(id)}
    >
      {children}
    </button>
  );
};

Tabs.Panel = function TabPanel({ id, children }) {
  const { activeTab } = useContext(TabsContext);
  return activeTab === id ? <div>{children}</div> : null;
};

// Usage
<Tabs defaultTab="tab1">
  <Tabs.List>
    <Tabs.Tab id="tab1">Tab 1</Tabs.Tab>
    <Tabs.Tab id="tab2">Tab 2</Tabs.Tab>
  </Tabs.List>
  <Tabs.Panel id="tab1">Content 1</Tabs.Panel>
  <Tabs.Panel id="tab2">Content 2</Tabs.Panel>
</Tabs>
```

### Render Props

```jsx
function MouseTracker({ render }) {
  const [position, setPosition] = useState({ x: 0, y: 0 });
  
  useEffect(() => {
    const handleMove = (e) => {
      setPosition({ x: e.clientX, y: e.clientY });
    };
    window.addEventListener('mousemove', handleMove);
    return () => window.removeEventListener('mousemove', handleMove);
  }, []);
  
  return render(position);
}

// Usage
<MouseTracker 
  render={({ x, y }) => (
    <div>Mouse position: {x}, {y}</div>
  )}
/>
```

## ✅ PropTypes (Runtime Validation)

```jsx
import PropTypes from 'prop-types';

function User({ name, age, email, role, onUpdate }) {
  // ...
}

User.propTypes = {
  name: PropTypes.string.isRequired,
  age: PropTypes.number,
  email: PropTypes.string.isRequired,
  role: PropTypes.oneOf(['user', 'admin', 'moderator']),
  onUpdate: PropTypes.func,
  
  // Complex types
  user: PropTypes.shape({
    id: PropTypes.number.isRequired,
    name: PropTypes.string
  }),
  
  items: PropTypes.arrayOf(PropTypes.string),
  
  // Custom validator
  customProp: function(props, propName, componentName) {
    if (!/^[A-Z]/.test(props[propName])) {
      return new Error('Must start with uppercase');
    }
  }
};

User.defaultProps = {
  age: 0,
  role: 'user'
};
```

## 🎯 Summary

- Components should have **single responsibility**
- **Props** pass data down from parent to child
- **Children** prop allows component composition
- Use **destructuring** for cleaner prop handling
- **Spread** props to pass through to underlying elements
- Consider **patterns** like container/presentational
