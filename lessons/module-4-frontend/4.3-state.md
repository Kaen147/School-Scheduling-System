## 📖 What is State?

State is data that changes over time and affects what's rendered.

```jsx
// Without state - static
function Counter() {
  const count = 0;
  return <p>Count: {count}</p>;  // Always shows 0
}

// With state - dynamic
function Counter() {
  const [count, setCount] = useState(0);
  return (
    <div>
      <p>Count: {count}</p>
      <button onClick={() => setCount(count + 1)}>+</button>
    </div>
  );
}
```

## 🔧 useState Hook

```jsx
import { useState } from 'react';

function Example() {
  // Syntax: const [value, setValue] = useState(initialValue);
  const [count, setCount] = useState(0);
  const [name, setName] = useState('');
  const [isOpen, setIsOpen] = useState(false);
  const [items, setItems] = useState([]);
  const [user, setUser] = useState(null);
  
  return (/* ... */);
}
```

### Lazy Initialization

For expensive initial values:

```jsx
// ❌ Runs on every render
const [data, setData] = useState(expensiveCalculation());

// ✅ Runs only on first render
const [data, setData] = useState(() => expensiveCalculation());
```

## 📊 Different State Types

### Primitive State

```jsx
function PrimitiveExamples() {
  const [count, setCount] = useState(0);
  const [text, setText] = useState('');
  const [isActive, setIsActive] = useState(false);
  
  return (
    <div>
      <button onClick={() => setCount(count + 1)}>
        Count: {count}
      </button>
      
      <input 
        value={text}
        onChange={(e) => setText(e.target.value)}
      />
      
      <button onClick={() => setIsActive(!isActive)}>
        {isActive ? 'Active' : 'Inactive'}
      </button>
    </div>
  );
}
```

### Object State

```jsx
function ObjectState() {
  const [user, setUser] = useState({
    name: '',
    email: '',
    age: 0
  });
  
  // ❌ Wrong - mutates state directly
  const updateNameWrong = () => {
    user.name = 'John';  // Don't do this!
    setUser(user);
  };
  
  // ✅ Correct - create new object
  const updateName = (name) => {
    setUser({ ...user, name });
  };
  
  // ✅ Update multiple fields
  const updateUser = (updates) => {
    setUser(prev => ({ ...prev, ...updates }));
  };
  
  return (
    <form>
      <input
        value={user.name}
        onChange={(e) => updateUser({ name: e.target.value })}
      />
      <input
        value={user.email}
        onChange={(e) => updateUser({ email: e.target.value })}
      />
    </form>
  );
}
```

### Array State

```jsx
function ArrayState() {
  const [items, setItems] = useState(['Apple', 'Banana']);
  
  // Add item
  const addItem = (item) => {
    setItems([...items, item]);
  };
  
  // Remove item
  const removeItem = (index) => {
    setItems(items.filter((_, i) => i !== index));
  };
  
  // Update item
  const updateItem = (index, newValue) => {
    setItems(items.map((item, i) => 
      i === index ? newValue : item
    ));
  };
  
  // Insert at position
  const insertAt = (index, item) => {
    setItems([
      ...items.slice(0, index),
      item,
      ...items.slice(index)
    ]);
  };
  
  return (
    <ul>
      {items.map((item, index) => (
        <li key={index}>
          {item}
          <button onClick={() => removeItem(index)}>×</button>
        </li>
      ))}
    </ul>
  );
}
```

### Nested Object State

```jsx
function NestedState() {
  const [form, setForm] = useState({
    personal: {
      name: '',
      email: ''
    },
    address: {
      street: '',
      city: ''
    }
  });
  
  // Update nested field
  const updatePersonal = (field, value) => {
    setForm(prev => ({
      ...prev,
      personal: {
        ...prev.personal,
        [field]: value
      }
    }));
  };
  
  return (
    <input
      value={form.personal.name}
      onChange={(e) => updatePersonal('name', e.target.value)}
    />
  );
}
```

## 🔄 Functional Updates

When new state depends on previous state:

```jsx
function Counter() {
  const [count, setCount] = useState(0);
  
  // ❌ May not work correctly with rapid clicks
  const incrementWrong = () => {
    setCount(count + 1);
    setCount(count + 1);  // Still uses old count!
  };
  
  // ✅ Always uses latest state
  const incrementCorrect = () => {
    setCount(prev => prev + 1);
    setCount(prev => prev + 1);  // Works correctly!
  };
  
  // ✅ Use functional update when depending on previous state
  const increment = () => setCount(prev => prev + 1);
  const decrement = () => setCount(prev => prev - 1);
  const reset = () => setCount(0);  // Not dependent, direct value OK
  
  return (
    <div>
      <p>{count}</p>
      <button onClick={increment}>+</button>
      <button onClick={decrement}>-</button>
      <button onClick={reset}>Reset</button>
    </div>
  );
}
```

## 🎯 State Management Patterns

### Lifting State Up

When siblings need to share state:

```jsx
// Parent holds the shared state
function Parent() {
  const [value, setValue] = useState('');
  
  return (
    <div>
      <Input value={value} onChange={setValue} />
      <Display value={value} />
    </div>
  );
}

function Input({ value, onChange }) {
  return (
    <input 
      value={value}
      onChange={(e) => onChange(e.target.value)}
    />
  );
}

function Display({ value }) {
  return <p>You typed: {value}</p>;
}
```

### Derived State

Calculate from existing state instead of storing:

```jsx
function ShoppingCart() {
  const [items, setItems] = useState([
    { id: 1, name: 'Apple', price: 1.00, quantity: 3 },
    { id: 2, name: 'Banana', price: 0.50, quantity: 5 }
  ]);
  
  // ❌ Don't store derived values
  // const [total, setTotal] = useState(0);
  
  // ✅ Calculate from state
  const total = items.reduce(
    (sum, item) => sum + item.price * item.quantity, 
    0
  );
  
  const itemCount = items.reduce(
    (sum, item) => sum + item.quantity, 
    0
  );
  
  return (
    <div>
      <p>Items: {itemCount}</p>
      <p>Total: ${total.toFixed(2)}</p>
    </div>
  );
}
```

### Controlled vs Uncontrolled

```jsx
// Controlled - React manages value
function ControlledInput() {
  const [value, setValue] = useState('');
  
  return (
    <input 
      value={value}
      onChange={(e) => setValue(e.target.value)}
    />
  );
}

// Uncontrolled - DOM manages value
function UncontrolledInput() {
  const inputRef = useRef();
  
  const handleSubmit = () => {
    console.log(inputRef.current.value);
  };
  
  return <input ref={inputRef} defaultValue="" />;
}
```

## ⚠️ Common Mistakes

### 1. Mutating State Directly

```jsx
// ❌ Wrong
const [user, setUser] = useState({ name: 'John' });
user.name = 'Jane';  // Mutation!
setUser(user);

// ✅ Correct
setUser({ ...user, name: 'Jane' });
```

### 2. Using Stale State in Callbacks

```jsx
function Timer() {
  const [count, setCount] = useState(0);
  
  useEffect(() => {
    const interval = setInterval(() => {
      // ❌ Always uses initial count (0)
      // setCount(count + 1);
      
      // ✅ Uses latest count
      setCount(prev => prev + 1);
    }, 1000);
    
    return () => clearInterval(interval);
  }, []);
  
  return <p>{count}</p>;
}
```

### 3. Storing Derived Data

```jsx
// ❌ Redundant state
const [items, setItems] = useState([]);
const [filteredItems, setFilteredItems] = useState([]);
const [itemCount, setItemCount] = useState(0);

// ✅ Derive from source state
const [items, setItems] = useState([]);
const [filter, setFilter] = useState('');

const filteredItems = items.filter(item => 
  item.name.includes(filter)
);
const itemCount = items.length;
```

## 🎯 Summary

- **State** is data that changes and triggers re-renders
- **useState** returns [value, setter] pair
- **Never mutate** state directly
- Use **functional updates** when depending on previous state
- **Lift state up** when siblings need to share
- **Derive** values instead of storing redundant state
