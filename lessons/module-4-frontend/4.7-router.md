## 📖 What is Client-Side Routing?

Traditional websites reload the entire page when navigating. Single Page Applications (SPAs) update only the content that changes, making navigation instant.

React Router handles this by:
1. Intercepting link clicks
2. Updating the URL without page reload
3. Rendering the appropriate component

## 🔗 Navigation with Link

Never use `<a>` tags for internal navigation - they cause full page reloads!

```jsx
import { Link, NavLink } from 'react-router-dom';

function Navbar() {
  return (
    <nav>
      {/* Basic Link */}
      <Link to="/">Home</Link>
      <Link to="/about">About</Link>
      
      {/* NavLink - adds "active" class when route matches */}
      <NavLink 
        to="/contact"
        className={({ isActive }) => isActive ? 'active' : ''}
      >
        Contact
      </NavLink>
    </nav>
  );
}
```

### Styling Active Links

```css
/* NavLink automatically gets "active" class */
nav a.active {
  color: blue;
  font-weight: bold;
}
```

## 🎯 Route Parameters

Dynamic segments in URLs:

```jsx
// App.jsx
<Routes>
  <Route path="/users" element={<UserList />} />
  <Route path="/users/:userId" element={<UserProfile />} />
  <Route path="/posts/:postId/comments/:commentId" element={<Comment />} />
</Routes>
```

### Accessing Parameters

```jsx
import { useParams } from 'react-router-dom';

function UserProfile() {
  const { userId } = useParams();
  const [user, setUser] = useState(null);
  
  useEffect(() => {
    // Fetch user with userId
    api.get(`/users/${userId}`)
      .then(res => setUser(res.data));
  }, [userId]);
  
  if (!user) return <p>Loading...</p>;
  
  return (
    <div>
      <h1>{user.name}</h1>
      <p>{user.email}</p>
    </div>
  );
}
```

## 🔍 Query Strings

Handle URL parameters like `/search?q=react&page=2`:

```jsx
import { useSearchParams } from 'react-router-dom';

function SearchPage() {
  const [searchParams, setSearchParams] = useSearchParams();
  
  // Read query params
  const query = searchParams.get('q') || '';
  const page = parseInt(searchParams.get('page')) || 1;
  
  // Update query params
  const handleSearch = (newQuery) => {
    setSearchParams({ q: newQuery, page: 1 });
  };
  
  const handlePageChange = (newPage) => {
    setSearchParams({ q: query, page: newPage });
  };
  
  return (
    <div>
      <input 
        value={query}
        onChange={(e) => handleSearch(e.target.value)}
        placeholder="Search..."
      />
      <p>Searching for: {query}</p>
      <p>Page: {page}</p>
      <button onClick={() => handlePageChange(page + 1)}>
        Next Page
      </button>
    </div>
  );
}
```

## 🧭 Programmatic Navigation

Navigate from code (after form submit, etc.):

```jsx
import { useNavigate } from 'react-router-dom';

function LoginForm() {
  const navigate = useNavigate();
  
  const handleSubmit = async (e) => {
    e.preventDefault();
    
    try {
      await api.post('/login', { email, password });
      
      // Navigate to dashboard after login
      navigate('/dashboard');
      
      // Or replace history (can't go back)
      navigate('/dashboard', { replace: true });
      
      // Go back
      navigate(-1);
      
      // Go forward
      navigate(1);
    } catch (error) {
      setError('Login failed');
    }
  };
  
  return <form onSubmit={handleSubmit}>...</form>;
}
```

## 🔒 Protected Routes

Restrict access to authenticated users:

```jsx
import { Navigate, useLocation } from 'react-router-dom';

function ProtectedRoute({ children }) {
  const token = localStorage.getItem('token');
  const location = useLocation();
  
  if (!token) {
    // Redirect to login, save attempted URL
    return <Navigate to="/login" state={{ from: location }} replace />;
  }
  
  return children;
}

// Usage in App.jsx
<Routes>
  <Route path="/login" element={<Login />} />
  <Route 
    path="/dashboard" 
    element={
      <ProtectedRoute>
        <Dashboard />
      </ProtectedRoute>
    } 
  />
</Routes>
```

### Redirect After Login

```jsx
function Login() {
  const navigate = useNavigate();
  const location = useLocation();
  
  // Get the page user tried to visit
  const from = location.state?.from?.pathname || '/dashboard';
  
  const handleLogin = async (credentials) => {
    await api.post('/login', credentials);
```

## 🚫 404 Not Found

Handle unknown routes:

```jsx
<Routes>
  <Route path="/" element={<Home />} />
  <Route path="/about" element={<About />} />
  
  {/* Catch all unmatched routes */}
  <Route path="*" element={<NotFound />} />
</Routes>
```

```jsx
function NotFound() {
  return (
    <div>
      <h1>404 - Page Not Found</h1>
      <p>The page you're looking for doesn't exist.</p>
      <Link to="/">Go Home</Link>
    </div>
  );
}
```

## 📱 Complete Example

```jsx
// App.jsx
import { Routes, Route } from 'react-router-dom';
import Layout from './components/Layout';
import Home from './pages/Home';
import Login from './pages/Login';
import Dashboard from './pages/Dashboard';
import UserProfile from './pages/UserProfile';
import NotFound from './pages/NotFound';
import ProtectedRoute from './components/ProtectedRoute';

function App() {
  return (
    <Routes>
      <Route path="/" element={<Layout />}>
        {/* Public routes */}
        <Route index element={<Home />} />
        <Route path="login" element={<Login />} />
        
        {/* Protected routes */}
        <Route 
          path="dashboard" 
          element={
            <ProtectedRoute>
              <Dashboard />
            </ProtectedRoute>
          } 
        />
        <Route 
          path="users/:userId" 
          element={
            <ProtectedRoute>
              <UserProfile />
            </ProtectedRoute>
          } 
        />
        
        {/* 404 */}
        <Route path="*" element={<NotFound />} />
      </Route>
    </Routes>
  );
}
```

## 🎯 Summary

- **BrowserRouter** wraps your app for routing
- **Routes/Route** define which component renders for each path
- **Link/NavLink** for navigation without page reload
- **useParams** accesses URL parameters
- **useSearchParams** handles query strings
- **useNavigate** for programmatic navigation
- **Outlet** renders nested route children
- **Protected routes** restrict access to authenticated users
