## 📖 Creating Classes

### Basic Class Syntax

```javascript
class Person {
  // Constructor - runs when creating new instance
  constructor(name, age) {
    this.name = name;
    this.age = age;
  }
  
  // Method
  greet() {
    return `Hello, I'm ${this.name}!`;
  }
  
  // Another method
  haveBirthday() {
    this.age++;
    return `Happy birthday! Now ${this.age} years old.`;
  }
}

// Create instances (objects)
const john = new Person("John", 25);
const jane = new Person("Jane", 30);

console.log(john.greet());        // "Hello, I'm John!"
console.log(jane.haveBirthday()); // "Happy birthday! Now 31 years old."
```

### Class Properties

```javascript
class Product {
  // Class field (property without constructor)
  currency = "USD";
  
  constructor(name, price) {
    this.name = name;
    this.price = price;
  }
  
  getFormattedPrice() {
    return `${this.currency} ${this.price.toFixed(2)}`;
  }
}

const laptop = new Product("Laptop", 999.99);
console.log(laptop.currency);           // "USD"
console.log(laptop.getFormattedPrice()); // "USD 999.99"
```

## 🔧 Constructor Deep Dive

### Default Parameter Values

class User {

### Validation in Constructor

```javascript
class BankAccount {
  constructor(owner, initialBalance = 0) {
    if (!owner || typeof owner !== "string") {
      throw new Error("Owner name is required");
    }
    if (initialBalance < 0) {
      throw new Error("Initial balance cannot be negative");
    }
    
    this.owner = owner;
    this.balance = initialBalance;
    this.accountNumber = this.#generateAccountNumber();
  }
  
  #generateAccountNumber() {
    return Math.random().toString(36).substr(2, 9).toUpperCase();
  }
}

try {
  const account = new BankAccount("", 100);  // Throws error
} catch (error) {
  console.error(error.message);  // "Owner name is required"
}
```

## 📊 Methods

### Instance Methods

Methods that operate on instance data.

```javascript
class Calculator {
  constructor(initialValue = 0) {
    this.value = initialValue;
  }
  
  add(n) {
    this.value += n;
    return this;  // Enable chaining
  }
  
  subtract(n) {
    this.value -= n;
    return this;
  }
  
  multiply(n) {
    this.value *= n;
    return this;
  }
  
  divide(n) {
    if (n === 0) throw new Error("Cannot divide by zero");
    this.value /= n;
    return this;
  }
  
  getResult() {
    return this.value;
  }
  
  reset() {
    this.value = 0;
    return this;
  }
}

// Method chaining
const result = new Calculator(10)
  .add(5)
  .multiply(2)
  .subtract(10)
  .getResult();

console.log(result);  // 20
```

### Static Methods

Methods that belong to the class itself, not instances.

```javascript
class MathUtils {
  // Static methods - called on the class, not instances
  static add(a, b) {
    return a + b;
  }
  
  static multiply(a, b) {
    return a * b;
  }
  
  static isEven(n) {
    return n % 2 === 0;
  }
  
  static randomBetween(min, max) {
    return Math.floor(Math.random() * (max - min + 1)) + min;
  }
}

// Called on the class directly
console.log(MathUtils.add(5, 3));           // 8
console.log(MathUtils.isEven(4));           // true
console.log(MathUtils.randomBetween(1, 10)); // Random 1-10

// NOT on instances
const utils = new MathUtils();
// utils.add(5, 3);  // Error! add is not a function
```

### Static Properties

```javascript
class User {
  static count = 0;
  static roles = ["user", "admin", "moderator"];
  
  constructor(name, role = "user") {
    this.name = name;
    this.role = role;
    this.id = ++User.count;  // Access static via class name
  }
  
  static getCount() {
    return User.count;
  }
  
  static isValidRole(role) {
    return User.roles.includes(role);
  }
}

const user1 = new User("John");
const user2 = new User("Jane");
const user3 = new User("Bob");

console.log(User.count);       // 3
console.log(User.getCount());  // 3
console.log(User.isValidRole("admin"));  // true
```

### When to Use Static

```javascript
// Use static for:
// 1. Utility functions that don't need instance data
class StringUtils {
  static capitalize(str) {
    return str.charAt(0).toUpperCase() + str.slice(1);
  }
}

// 2. Factory methods
class User {
  constructor(name, email) {
    this.name = name;
    this.email = email;
  }
  
  static fromJSON(json) {
    const data = JSON.parse(json);
    return new User(data.name, data.email);
  }
  
  static createGuest() {
    return new User("Guest", "guest@example.com");
  }
}

const user = User.fromJSON('{"name":"John","email":"john@example.com"}');
const guest = User.createGuest();

// 3. Tracking class-level data
class Order {
  static orderCount = 0;
  
  constructor(items) {
    this.id = ++Order.orderCount;
    this.items = items;
  }
}
```

## 🔐 Getters and Setters

Control how properties are accessed and modified.

### Basic Getters and Setters

```javascript
class Circle {
  constructor(radius) {
    this._radius = radius;  // Convention: _ prefix for "private"
  }
  
  // Getter - accessed like a property
  get radius() {
    return this._radius;
  }
  
  // Setter - assigned like a property
  set radius(value) {
    if (value <= 0) {
      throw new Error("Radius must be positive");
    }
    this._radius = value;
  }
  
  // Computed property (getter only)
  get diameter() {
    return this._radius * 2;
  }
  
  get area() {
    return Math.PI * this._radius ** 2;
  }
  
  get circumference() {
    return 2 * Math.PI * this._radius;
  }
}

const circle = new Circle(5);

// Access like properties (not methods!)
console.log(circle.radius);        // 5
console.log(circle.diameter);      // 10
console.log(circle.area);          // 78.54...
console.log(circle.circumference); // 31.42...

// Set like property
circle.radius = 10;
console.log(circle.area);  // 314.16...

// Validation works
// circle.radius = -5;  // Error: Radius must be positive
```

### Practical Getter/Setter Examples

```javascript
class User {
  constructor(firstName, lastName, birthYear) {
    this.firstName = firstName;
    this.lastName = lastName;
    this.birthYear = birthYear;
  }
  
  // Computed full name
  get fullName() {
    return `${this.firstName} ${this.lastName}`;
  }
  
  // Set full name by splitting
  set fullName(value) {
    const parts = value.split(" ");
    this.firstName = parts[0];
    this.lastName = parts.slice(1).join(" ");
  }
  
  // Computed age
  get age() {
    return new Date().getFullYear() - this.birthYear;
  }
}

const user = new User("John", "Doe", 1990);
console.log(user.fullName);  // "John Doe"
console.log(user.age);       // 36 (in 2026)

user.fullName = "Jane Smith";
console.log(user.firstName); // "Jane"
console.log(user.lastName);  // "Smith"
```

## 🎯 The `this` Keyword

`this` refers to the object that is executing the current function.

### this in Methods

```javascript
class Counter {
  constructor() {
    this.count = 0;
  }
  
  increment() {
    this.count++;  // 'this' is the counter instance
    console.log(this.count);
  }
}

const counter = new Counter();
counter.increment();  // 1 - 'this' is counter
counter.increment();  // 2
```

### this Problem with Callbacks

```javascript
class Timer {
  constructor() {
    this.seconds = 0;
  }
  
  start() {
    // PROBLEM: 'this' is lost in callback
    setInterval(function() {
      this.seconds++;  // 'this' is undefined or window!
      console.log(this.seconds);
    }, 1000);
  }
}

// Solutions:

class Timer {
  constructor() {
    this.seconds = 0;
  }
  
  // Solution 1: Arrow function (inherits 'this')
  start() {
    setInterval(() => {
      this.seconds++;  // 'this' is the Timer instance
      console.log(this.seconds);
    }, 1000);
  }
  
  // Solution 2: Bind 'this'
  startWithBind() {
    setInterval(function() {
      this.seconds++;
      console.log(this.seconds);
    }.bind(this), 1000);
  }
  
  // Solution 3: Save reference
  startWithRef() {
    const self = this;
    setInterval(function() {
      self.seconds++;
      console.log(self.seconds);
    }, 1000);
  }
}
```

### Arrow Functions and this

```javascript
class Button {
  constructor(label) {
    this.label = label;
    this.clicks = 0;
  }
  
  // Regular method - 'this' depends on how it's called
  handleClick() {
    this.clicks++;
    console.log(`${this.label} clicked ${this.clicks} times`);
  }
  
  // Arrow function property - 'this' is always the instance
  handleClickArrow = () => {
    this.clicks++;
    console.log(`${this.label} clicked ${this.clicks} times`);
  }
}

const button = new Button("Submit");

// Regular method - works when called on object
button.handleClick();  // Works

// But fails when passed as callback
// document.addEventListener("click", button.handleClick);  // 'this' is lost!

// Arrow function always works
// document.addEventListener("click", button.handleClickArrow);  // Works!
```

## 🏭 Factory Pattern

Alternative to classes for creating objects.

```javascript
// Factory function
function createUser(name, email) {
  // Private variables (closure)
  let loginCount = 0;
  
  return {
    name,
    email,
    
    login() {
      loginCount++;
      console.log(`${name} logged in. Total: ${loginCount}`);
    },
    
    getLoginCount() {
      return loginCount;
    }
  };
}

const user = createUser("John", "john@example.com");
user.login();  // "John logged in. Total: 1"
user.login();  // "John logged in. Total: 2"
console.log(user.loginCount);  // undefined - truly private!
```

### Class vs Factory

```javascript
// Class approach
class User {
  #loginCount = 0;  // Private field
  
  constructor(name, email) {
    this.name = name;
    this.email = email;
  }
  
  login() {
    this.#loginCount++;
  }
}

// Factory approach
function createUser(name, email) {
  let loginCount = 0;
  
  return {
    name,
    email,
    login() { loginCount++; }
  };
}

// Both achieve similar results
// Classes: better for inheritance, instanceof checks
// Factories: simpler, true privacy, no 'this' issues
```

## 🎯 Summary

- **Classes** are blueprints for creating objects
- **Constructor** initializes new instances
- **Instance methods** operate on instance data
- **Static methods** belong to the class itself
- **Getters/Setters** control property access
- **this** refers to the current instance (be careful with callbacks!)
- Use **arrow functions** or **bind** to preserve `this` in callbacks
- **Factory functions** are an alternative to classes
