## 📖 What is Polymorphism?

Polymorphism means "many forms." It allows objects of different classes to be treated as objects of a common parent class, while each responds differently to the same method call.

### Simple Example

```javascript
class Animal {
  speak() {
    return "Some sound";
  }
}

class Dog extends Animal {
  speak() {
    return "Woof!";
  }
}

class Cat extends Animal {
  speak() {
    return "Meow!";
  }
}

class Cow extends Animal {
  speak() {
    return "Moo!";
  }
}

// Polymorphism in action - same method, different behavior
const animals = [new Dog(), new Cat(), new Cow()];

animals.forEach(animal => {
  console.log(animal.speak());
});
// "Woof!"
// "Meow!"
// "Moo!"
```

### Why It Matters

```javascript
// Without polymorphism - checking types everywhere
function makeSound(animal) {
  if (animal instanceof Dog) {
    console.log("Woof!");
  } else if (animal instanceof Cat) {
    console.log("Meow!");
  } else if (animal instanceof Cow) {
    console.log("Moo!");
  }
  // Need to add more conditions for each new animal!
}

// With polymorphism - clean and extensible
function makeSound(animal) {
  console.log(animal.speak());
  // Works with ANY animal, even ones not yet created!
}
```

## 🦆 Duck Typing

JavaScript uses "duck typing" - if it walks like a duck and quacks like a duck, it's a duck.

```javascript
// These don't need to share a parent class
class Duck {
  quack() {
    return "Quack!";
  }
  swim() {
    return "Swimming...";
  }
}

class Person {
  quack() {
    return "I'm pretending to quack!";
  }
  swim() {
    return "Swimming...";
  }
}

class RubberDuck {
  quack() {
    return "Squeak!";
  }
  swim() {
    return "Floating...";
  }
}

// Function doesn't care about the type, just the interface
function duckTest(duck) {
  console.log(duck.quack());
  console.log(duck.swim());
}

duckTest(new Duck());       // Works!
duckTest(new Person());     // Works!
duckTest(new RubberDuck()); // Works!
```

## 🎨 Practical Polymorphism Examples

### Payment Processing

```javascript
class PaymentProcessor {
  process(amount) {
    throw new Error("Subclass must implement process()");
  }
  
  refund(transactionId) {
    throw new Error("Subclass must implement refund()");
  }
}

class CreditCardProcessor extends PaymentProcessor {
  process(amount) {
    console.log(`Processing $${amount} via Credit Card`);
    return { success: true, transactionId: `CC-${Date.now()}` };
  }
  
  refund(transactionId) {
    console.log(`Refunding transaction ${transactionId} to card`);
    return { success: true };
  }
}

class PayPalProcessor extends PaymentProcessor {
  process(amount) {
    console.log(`Processing $${amount} via PayPal`);
    return { success: true, transactionId: `PP-${Date.now()}` };
  }
  
  refund(transactionId) {
    console.log(`Refunding transaction ${transactionId} to PayPal`);
    return { success: true };
  }
}

class CryptoProcessor extends PaymentProcessor {
  process(amount) {
    console.log(`Processing $${amount} via Cryptocurrency`);
    return { success: true, transactionId: `CRYPTO-${Date.now()}` };
  }
  
  refund(transactionId) {
    console.log(`Crypto refunds not supported`);
    return { success: false, reason: "Crypto payments are non-refundable" };
  }
}

// Checkout system doesn't care which processor is used
class Checkout {
  constructor(paymentProcessor) {
    this.processor = paymentProcessor;
  }
  
  completePurchase(amount) {
    return this.processor.process(amount);
  }
  
  handleRefund(transactionId) {
    return this.processor.refund(transactionId);
  }
}

// Easy to switch payment methods
const checkout1 = new Checkout(new CreditCardProcessor());
const checkout2 = new Checkout(new PayPalProcessor());
const checkout3 = new Checkout(new CryptoProcessor());

checkout1.completePurchase(99.99);
checkout2.completePurchase(49.99);
checkout3.completePurchase(199.99);
```

### Notification System

```javascript
class Notifier {
  send(message, recipient) {
    throw new Error("Subclass must implement send()");
  }
}

class EmailNotifier extends Notifier {
  send(message, recipient) {
    console.log(`Sending email to ${recipient}: ${message}`);
    return { type: "email", status: "sent" };
  }
}

class SMSNotifier extends Notifier {
  send(message, recipient) {
    console.log(`Sending SMS to ${recipient}: ${message}`);
    return { type: "sms", status: "sent" };
  }
}

class PushNotifier extends Notifier {
  send(message, recipient) {
    console.log(`Sending push notification to ${recipient}: ${message}`);
    return { type: "push", status: "sent" };
  }
}

class SlackNotifier extends Notifier {
  send(message, recipient) {
    console.log(`Sending Slack message to ${recipient}: ${message}`);
    return { type: "slack", status: "sent" };
  }
}

// Notification service can use any notifier
class NotificationService {
  constructor(notifiers = []) {
    this.notifiers = notifiers;
  }
  
  addNotifier(notifier) {
    this.notifiers.push(notifier);
  }
  
  notifyAll(message, recipient) {
    return this.notifiers.map(notifier => 
      notifier.send(message, recipient)
    );
  }
}

const service = new NotificationService([
  new EmailNotifier(),
  new PushNotifier()
]);

service.addNotifier(new SlackNotifier());
service.notifyAll("Your order has shipped!", "user@example.com");
```

### Shape Rendering

```javascript
class Shape {
  draw(canvas) {
    throw new Error("Subclass must implement draw()");
  }
  
  area() {
    throw new Error("Subclass must implement area()");
  }
}

class Rectangle extends Shape {
  constructor(x, y, width, height, color) {
    super();
    this.x = x;
    this.y = y;
    this.width = width;
    this.height = height;
    this.color = color;
  }
  
  draw(canvas) {
    console.log(`Drawing ${this.color} rectangle at (${this.x}, ${this.y})`);
    // canvas.fillRect(this.x, this.y, this.width, this.height);
  }
  
  area() {
    return this.width * this.height;
  }
}

class Circle extends Shape {
  constructor(x, y, radius, color) {
    super();
    this.x = x;
    this.y = y;
    this.radius = radius;
    this.color = color;
  }
  
  draw(canvas) {
    console.log(`Drawing ${this.color} circle at (${this.x}, ${this.y})`);
    // canvas.arc(this.x, this.y, this.radius, 0, Math.PI * 2);
  }
  
  area() {
    return Math.PI * this.radius ** 2;
  }
}

class Triangle extends Shape {
  constructor(points, color) {
    super();
    this.points = points;  // [{x, y}, {x, y}, {x, y}]
    this.color = color;
  }
  
  draw(canvas) {
    console.log(`Drawing ${this.color} triangle`);
    // Draw triangle using points
  }
  
  area() {
    const [p1, p2, p3] = this.points;
    return Math.abs(
      (p1.x * (p2.y - p3.y) + p2.x * (p3.y - p1.y) + p3.x * (p1.y - p2.y)) / 2
    );
  }
}

// Renderer works with any shape
class Renderer {
  constructor(canvas) {
    this.canvas = canvas;
    this.shapes = [];
  }
  
  addShape(shape) {
    this.shapes.push(shape);
  }
  
  render() {
    this.shapes.forEach(shape => shape.draw(this.canvas));
  }
  
  getTotalArea() {
    return this.shapes.reduce((sum, shape) => sum + shape.area(), 0);
  }
}

const renderer = new Renderer("canvas");
renderer.addShape(new Rectangle(0, 0, 100, 50, "red"));
renderer.addShape(new Circle(200, 200, 30, "blue"));
renderer.addShape(new Triangle([{x:0,y:0}, {x:50,y:100}, {x:100,y:0}], "green"));

renderer.render();
console.log(`Total area: ${renderer.getTotalArea()}`);
```

## 🔌 Interface-Like Patterns

JavaScript doesn't have interfaces, but we can simulate them:

### Using Base Classes

```javascript
class Serializable {
  serialize() {
    throw new Error("Must implement serialize()");
  }
  
  static deserialize(data) {
    throw new Error("Must implement deserialize()");
  }
}

class User extends Serializable {
  constructor(name, email) {
    super();
    this.name = name;
    this.email = email;
  }
  
  serialize() {
    return JSON.stringify({ name: this.name, email: this.email });
  }
  
  static deserialize(data) {
    const { name, email } = JSON.parse(data);
    return new User(name, email);
  }
}
```

### Using Validation

```javascript
function validateInterface(obj, requiredMethods) {
  for (const method of requiredMethods) {
    if (typeof obj[method] !== "function") {
      throw new Error(`Object must implement ${method}()`);
    }
  }
  return true;
}

// Usage
const storageInterface = ["save", "load", "delete"];

class LocalStorage {
  save(key, data) { /* ... */ }
  load(key) { /* ... */ }
  delete(key) { /* ... */ }
}

class CloudStorage {
  save(key, data) { /* ... */ }
  load(key) { /* ... */ }
  delete(key) { /* ... */ }
}

function useStorage(storage) {
  validateInterface(storage, storageInterface);
  // Now safe to use storage.save(), storage.load(), etc.
}
```

## 🎯 Summary

- **Polymorphism** = same interface, different implementations
- Objects respond differently to the same method call
- **Duck typing** - behavior matters more than type
- Design for **interfaces**, not implementations
- Makes code **extensible** - add new types without changing existing code
- Reduces **conditional logic** (no more instanceof checks)
