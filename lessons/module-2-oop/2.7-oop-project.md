## 📋 Requirements

### Entities
- **Book** - title, author, ISBN, available copies
- **Member** - name, memberId, borrowed books, membership type
- **Library** - collection of books and members
- **Loan** - tracks who borrowed what and when

### Features
- Add/remove books
- Register/remove members
- Borrow and return books
- Search books by title, author, or ISBN
- Track overdue books
- Different member types with different borrowing limits

## 📚 Step 2: Book Class

```javascript
class Book extends Entity {
  #title;
  #author;
  #isbn;
  #totalCopies;
  #availableCopies;
  #category;
  
  constructor(title, author, isbn, copies = 1, category = "General") {
    super();
    this.#title = title;
    this.#author = author;
    this.#isbn = isbn;
    this.#totalCopies = copies;
    this.#availableCopies = copies;
    this.#category = category;
  }
  
  // Getters
  get title() { return this.#title; }
  get author() { return this.#author; }
  get isbn() { return this.#isbn; }
  get category() { return this.#category; }
  get totalCopies() { return this.#totalCopies; }
  get availableCopies() { return this.#availableCopies; }
  get isAvailable() { return this.#availableCopies > 0; }
  
  // Methods
  checkout() {
    if (!this.isAvailable) {
      throw new Error(`"${this.#title}" is not available`);
    }
    this.#availableCopies--;
    return true;
  }
  
  return() {
    if (this.#availableCopies >= this.#totalCopies) {
      throw new Error("All copies already returned");
    }
    this.#availableCopies++;
    return true;
  }
  
  addCopies(count) {
    if (count <= 0) throw new Error("Count must be positive");
    this.#totalCopies += count;
    this.#availableCopies += count;
  }
  
  matches(query) {
    const q = query.toLowerCase();
    return (
      this.#title.toLowerCase().includes(q) ||
      this.#author.toLowerCase().includes(q) ||
      this.#isbn.includes(q)
    );
  }
  
  toJSON() {
    return {
      id: this.id,
      title: this.#title,
      author: this.#author,
      isbn: this.#isbn,
      category: this.#category,
      totalCopies: this.#totalCopies,
      availableCopies: this.#availableCopies
    };
  }
  
  toString() {
    return `"${this.#title}" by ${this.#author} (${this.#availableCopies}/${this.#totalCopies} available)`;
  }
}
```

## 👤 Step 3: Member Classes (Inheritance & Polymorphism)

```javascript
// Base Member class
class Member extends Entity {
  #name;
  #email;
  #borrowedBooks;
  #membershipDate;
  
  constructor(name, email) {
    super();
    if (new.target === Member) {
      throw new Error("Member is abstract - use a specific member type");
    }
    this.#name = name;
    this.#email = email;
    this.#borrowedBooks = [];
    this.#membershipDate = new Date();
  }
  
  // Getters
  get name() { return this.#name; }
  get email() { return this.#email; }
  get borrowedBooks() { return [...this.#borrowedBooks]; }
  get borrowedCount() { return this.#borrowedBooks.length; }
  get membershipDate() { return this.#membershipDate; }
  
  // Abstract - must be implemented by subclasses
  get borrowLimit() {
    throw new Error("Subclass must implement borrowLimit");
  }
  
  get loanDuration() {
    throw new Error("Subclass must implement loanDuration");
  }
  
  get memberType() {
    throw new Error("Subclass must implement memberType");
  }
  
  // Methods
  canBorrow() {
    return this.#borrowedBooks.length < this.borrowLimit;
  }
  
  borrow(bookId) {
    if (!this.canBorrow()) {
      throw new Error(`Borrow limit (${this.borrowLimit}) reached`);
    }
    if (this.#borrowedBooks.includes(bookId)) {
      throw new Error("Already borrowed this book");
    }
    this.#borrowedBooks.push(bookId);
  }
  
  returnBook(bookId) {
    const index = this.#borrowedBooks.indexOf(bookId);
    if (index === -1) {
      throw new Error("This book was not borrowed by this member");
    }
    this.#borrowedBooks.splice(index, 1);
  }
  
  hasBorrowed(bookId) {
    return this.#borrowedBooks.includes(bookId);
  }
  
  toJSON() {
    return {
      id: this.id,
      name: this.#name,
      email: this.#email,
      memberType: this.memberType,
      borrowedBooks: this.#borrowedBooks,
      borrowLimit: this.borrowLimit
    };
  }
}

// Concrete member types - Polymorphism
```

## 📝 Step 4: Loan Class

```javascript
class Loan extends Entity {
  #bookId;
  #memberId;
  #borrowDate;
  #dueDate;
  #returnDate;
  #status;
  
  constructor(bookId, memberId, durationDays) {
    super();
    this.#bookId = bookId;
    this.#memberId = memberId;
    this.#borrowDate = new Date();
    this.#dueDate = new Date();
    this.#dueDate.setDate(this.#dueDate.getDate() + durationDays);
    this.#returnDate = null;
    this.#status = "active";
  }
  
  get bookId() { return this.#bookId; }
  get memberId() { return this.#memberId; }
  get borrowDate() { return this.#borrowDate; }
  get dueDate() { return this.#dueDate; }
  get returnDate() { return this.#returnDate; }
  get status() { return this.#status; }
  
  get isOverdue() {
    if (this.#status === "returned") return false;
    return new Date() > this.#dueDate;
  }
  
  get daysOverdue() {
    if (!this.isOverdue) return 0;
    const diff = new Date() - this.#dueDate;
    return Math.floor(diff / (1000 * 60 * 60 * 24));
  }
  
  complete() {
    this.#returnDate = new Date();
    this.#status = "returned";
  }
  
  toJSON() {
    return {
      id: this.id,
      bookId: this.#bookId,
      memberId: this.#memberId,
      borrowDate: this.#borrowDate,
      dueDate: this.#dueDate,
      returnDate: this.#returnDate,
      status: this.#status,
      isOverdue: this.isOverdue
    };
  }
}
```

## 🏛️ Step 5: Library Class (Facade)

```javascript
class Library {
  #name;
  #books;
  #members;
  #loans;
  
  constructor(name) {
    this.#name = name;
    this.#books = new Map();
    this.#members = new Map();
    this.#loans = [];
  }
  
  get name() { return this.#name; }
  get bookCount() { return this.#books.size; }
  get memberCount() { return this.#members.size; }
  get activeLoans() { return this.#loans.filter(l => l.status === "active").length; }
  
  // Book Management
  addBook(book) {
    if (!(book instanceof Book)) {
      throw new Error("Must be a Book instance");
    }
    this.#books.set(book.id, book);
    return book;
  }
  
  removeBook(bookId) {
    const book = this.#books.get(bookId);
    if (!book) throw new Error("Book not found");
    if (!book.isAvailable || book.availableCopies < book.totalCopies) {
      throw new Error("Cannot remove book with active loans");
    }
    this.#books.delete(bookId);
    return true;
  }
  
  getBook(bookId) {
    return this.#books.get(bookId);
  }
  
  searchBooks(query) {
    const results = [];
    for (const book of this.#books.values()) {
      if (book.matches(query)) {
        results.push(book);
      }
    }
    return results;
  }
  
  getAvailableBooks() {
    return [...this.#books.values()].filter(book => book.isAvailable);
  }
  
  getBooksByCategory(category) {
    return [...this.#books.values()].filter(book => book.category === category);
  }
  
  // Member Management
  registerMember(member) {
    if (!(member instanceof Member)) {
      throw new Error("Must be a Member instance");
    }
    this.#members.set(member.id, member);
    return member;
  }
  
  removeMember(memberId) {
    const member = this.#members.get(memberId);
    if (!member) throw new Error("Member not found");
    if (member.borrowedCount > 0) {
      throw new Error("Cannot remove member with borrowed books");
    }
    this.#members.delete(memberId);
    return true;
  }
  
  getMember(memberId) {
    return this.#members.get(memberId);
  }
  
  // Loan Management
  borrowBook(memberId, bookId) {
    const member = this.#members.get(memberId);
    const book = this.#books.get(bookId);
    
    if (!member) throw new Error("Member not found");
    if (!book) throw new Error("Book not found");
    if (!book.isAvailable) throw new Error("Book not available");
    if (!member.canBorrow()) throw new Error("Member has reached borrow limit");
    
    // Create loan
    const loan = new Loan(bookId, memberId, member.loanDuration);
    
    // Update book and member
    book.checkout();
    member.borrow(bookId);
    
    this.#loans.push(loan);
    
    return loan;
  }
  
  returnBook(memberId, bookId) {
    const member = this.#members.get(memberId);
    const book = this.#books.get(bookId);
    
    if (!member) throw new Error("Member not found");
    if (!book) throw new Error("Book not found");
    if (!member.hasBorrowed(bookId)) {
      throw new Error("This member did not borrow this book");
    }
    
    // Find active loan
    const loan = this.#loans.find(
      l => l.bookId === bookId && l.memberId === memberId && l.status === "active"
    );
    
    if (!loan) throw new Error("Loan not found");
    
    // Update everything
    loan.complete();
    book.return();
    member.returnBook(bookId);
    
    return {
      loan,
      wasOverdue: loan.isOverdue,
      daysOverdue: loan.daysOverdue
    };
  }
  
  getOverdueLoans() {
    return this.#loans.filter(loan => loan.isOverdue);
  }
  
  getMemberLoans(memberId) {
    return this.#loans.filter(loan => loan.memberId === memberId);
  }
  
  // Reports
  getStatistics() {
    const activeLoans = this.#loans.filter(l => l.status === "active");
    const overdueLoans = this.getOverdueLoans();
    
    return {
      totalBooks: this.#books.size,
      totalMembers: this.#members.size,
      activeLoans: activeLoans.length,
      overdueLoans: overdueLoans.length,
      availableBooks: this.getAvailableBooks().length
    };
  }
}
```
