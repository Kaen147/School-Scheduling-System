## 📖 What is Express?

Express is a minimal, flexible Node.js web framework that provides robust features for building web applications and APIs.

### Why Express?

- **Simple routing** - Easy to define URL handlers
- **Middleware** - Plug in functionality
- **Large ecosystem** - Many packages available
- **Industry standard** - Used by most Node.js apps

```bash
npm init -y
```

### Basic Server

```javascript
import express from 'express';

const app = express();
const PORT = 3000;

// Route handler
app.get('/', (req, res) => {
  res.send('Hello, World!');
});

// Start server
app.listen(PORT, () => {
  console.log(`Server running on http://localhost:${PORT}`);
});
```

## 🛤️ Routing

### Basic Routes

```javascript
// GET request
app.get('/users', (req, res) => {
  res.json([{ id: 1, name: 'John' }]);
});

// POST request
app.post('/users', (req, res) => {
  const newUser = req.body;
  res.status(201).json(newUser);
});

// PUT request
app.put('/users/:id', (req, res) => {
  const { id } = req.params;
  res.json({ id, ...req.body });
});

// DELETE request
app.delete('/users/:id', (req, res) => {
  res.status(204).send();
});

// Handle all methods
app.all('/secret', (req, res) => {
  res.send('Accessing secret section');
});
```

### Route Parameters

```javascript
// Single parameter
app.get('/users/:id', (req, res) => {
  const userId = req.params.id;
  res.json({ userId });
});

// Multiple parameters
app.get('/users/:userId/posts/:postId', (req, res) => {
  const { userId, postId } = req.params;
  res.json({ userId, postId });
});

// Optional parameter (use ? in route or check in handler)
app.get('/posts/:year/:month?', (req, res) => {
  const { year, month } = req.params;
  res.json({ year, month: month || 'all' });
});
```

### Query Parameters

```javascript
// URL: /search?q=javascript&page=2&limit=10
app.get('/search', (req, res) => {
  const { q, page = 1, limit = 10 } = req.query;
  res.json({ query: q, page, limit });
});
```

### Route Patterns

```javascript
// Wildcard
app.get('/files/*', (req, res) => {
  res.send(`File path: ${req.params[0]}`);
});

// Regex pattern
app.get(/.*fly$/, (req, res) => {
  res.send('Matches butterfly, dragonfly, etc.');
});
```

## 🔧 Request Object (req)

```javascript
app.post('/example', (req, res) => {
  // URL parameters
  req.params.id;
  
  // Query string
  req.query.search;
  
  // Request body (needs middleware)
  req.body.name;
  
  // Headers
  req.headers['content-type'];
  req.get('Authorization');
  
  // HTTP method
  req.method;  // 'POST'
  
  // URL path
  req.path;    // '/example'
  req.url;     // '/example?foo=bar'
  
  // IP address
  req.ip;
  
  // Cookies (needs cookie-parser)
  req.cookies.sessionId;
});
```

## 📤 Response Object (res)

```javascript
app.get('/responses', (req, res) => {
  // Send text
  res.send('Hello');
  
  // Send JSON
  res.json({ message: 'Hello' });
  
  // Send with status
  res.status(201).json({ created: true });
  
  // Send status only
  res.sendStatus(204);  // No Content
  
  // Redirect
  res.redirect('/new-location');
  res.redirect(301, '/permanent-redirect');
  
  // Send file
  res.sendFile('/path/to/file.pdf');
  
  // Download file
  res.download('/path/to/file.pdf', 'custom-name.pdf');
  
  // Set headers
  res.set('X-Custom-Header', 'value');
  res.setHeader('Content-Type', 'text/html');
  
  // Set cookie
  res.cookie('name', 'value', { maxAge: 900000, httpOnly: true });
  
  // Clear cookie
  res.clearCookie('name');
  
  // Render template (needs view engine)
  res.render('index', { title: 'Home' });
});
```

## 🔌 Middleware

Middleware functions have access to req, res, and next.

### Built-in Middleware

```javascript
// Parse JSON bodies
app.use(express.json());

// Parse URL-encoded bodies
app.use(express.urlencoded({ extended: true }));

// Serve static files
app.use(express.static('public'));
```

### Custom Middleware

```javascript
// Logger middleware
const logger = (req, res, next) => {
  console.log(`${new Date().toISOString()} - ${req.method} ${req.url}`);
  next();  // Pass to next middleware
};

app.use(logger);

// Auth middleware
const authenticate = (req, res, next) => {
  const token = req.headers.authorization;
  
  if (!token) {
    return res.status(401).json({ error: 'No token provided' });
  }
  
  // Verify token...
  req.user = { id: 1, name: 'John' };  // Attach user to request
  next();
};

// Use on specific routes
app.get('/protected', authenticate, (req, res) => {
  res.json({ user: req.user });
});
```

### Error Handling Middleware

```javascript
// Error handler (4 parameters)
const errorHandler = (err, req, res, next) => {
  console.error(err.stack);
  
  res.status(err.status || 500).json({
    error: {
      message: err.message || 'Internal Server Error'
    }
  });
};

// Must be last
app.use(errorHandler);

// Throwing errors in routes
app.get('/error', (req, res, next) => {
  const error = new Error('Something went wrong');
  error.status = 400;
  next(error);  // Pass to error handler
});

// Async error handling
app.get('/async-error', async (req, res, next) => {
  try {
    const data = await someAsyncOperation();
    res.json(data);
  } catch (error) {
    next(error);
  }
});
```

### Third-Party Middleware

```javascript
import cors from 'cors';
import helmet from 'helmet';
import morgan from 'morgan';

// CORS - Cross-Origin Resource Sharing
app.use(cors());

// Helmet - Security headers
app.use(helmet());

// Morgan - HTTP request logger
app.use(morgan('dev'));
```

## 📁 Router (Modular Routes)

```javascript
// routes/users.js
import { Router } from 'express';

const router = Router();

router.get('/', (req, res) => {
  res.json([{ id: 1, name: 'John' }]);
});

router.get('/:id', (req, res) => {
  res.json({ id: req.params.id });
});

router.post('/', (req, res) => {
  res.status(201).json(req.body);
});

export default router;

// server.js
import userRoutes from './routes/users.js';

app.use('/api/users', userRoutes);
// Routes become: /api/users, /api/users/:id, etc.
```

## 🎯 Summary

- **Express** simplifies Node.js server creation
- **Routes** handle different URLs and HTTP methods
- **Middleware** processes requests in a chain
- Use **req** for request data, **res** for responses
- **Router** helps organize routes into modules
- Always handle errors with error middleware
