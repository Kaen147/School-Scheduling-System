## 📖 What is MongoDB?

MongoDB is a NoSQL document database. Instead of tables and rows, it uses collections and documents (JSON-like objects).

### SQL vs MongoDB

| SQL | MongoDB |
|-----|---------|
| Database | Database |
| Table | Collection |
| Row | Document |
| Column | Field |

### Document Example

```javascript
{
  "_id": ObjectId("507f1f77bcf86cd799439011"),
  "name": "John Doe",
  "email": "john@example.com",
  "age": 25,
  "address": {
    "city": "New York",
    "country": "USA"
  },
  "hobbies": ["reading", "gaming"],
  "createdAt": ISODate("2024-01-15T10:30:00Z")
}
```

## 📋 Schemas & Models

### Basic Schema

```javascript
import mongoose from 'mongoose';

const userSchema = new mongoose.Schema({
  name: String,
  email: String,
  age: Number,
  isActive: Boolean,
  createdAt: Date
});

const User = mongoose.model('User', userSchema);
export default User;
```

### Schema with Options

```javascript
const userSchema = new mongoose.Schema({
  name: {
    type: String,
    required: [true, 'Name is required'],
    trim: true,
    minlength: [2, 'Name must be at least 2 characters'],
    maxlength: [50, 'Name cannot exceed 50 characters']
  },
  email: {
    type: String,
    required: true,
    unique: true,
    lowercase: true,
    match: [/^\S+@\S+\.\S+$/, 'Invalid email format']
  },
  password: {
    type: String,
    required: true,
    minlength: 8,
    select: false  // Don't include in queries by default
  },
  role: {
    type: String,
    enum: ['user', 'admin', 'moderator'],
    default: 'user'
  },
  age: {
    type: Number,
    min: [0, 'Age cannot be negative'],
    max: [150, 'Invalid age']
  },
  profile: {
    bio: String,
    avatar: String,
    website: String
  },
  tags: [String],
  friends: [{
    type: mongoose.Schema.Types.ObjectId,
    ref: 'User'
  }]
}, {
  timestamps: true  // Adds createdAt and updatedAt
});
```

### Schema Types

```javascript
const schema = new mongoose.Schema({
  // Basic types
  string: String,
  number: Number,
  boolean: Boolean,
  date: Date,
  buffer: Buffer,
  
  // ObjectId (for references)
  userId: mongoose.Schema.Types.ObjectId,
  
  // Mixed (any type)
  anything: mongoose.Schema.Types.Mixed,
  
  // Array of strings
  tags: [String],
  
  // Array of objects
  items: [{
    name: String,
    quantity: Number
  }],
  
  // Nested object
  address: {
    street: String,
    city: String,
    zip: String
  },
  
  // Map
  metadata: {
    type: Map,
    of: String
  }
});
```

## 📝 CRUD Operations

### Create

```javascript
// Method 1: new + save
const user = new User({
  name: 'John',
  email: 'john@example.com'
});
await user.save();

// Method 2: create
const user = await User.create({
  name: 'John',
  email: 'john@example.com'
});

// Create multiple
const users = await User.insertMany([
  { name: 'John', email: 'john@example.com' },
  { name: 'Jane', email: 'jane@example.com' }
]);
```

### Read

```javascript
// Find all
const users = await User.find();

// Find with conditions
const activeUsers = await User.find({ isActive: true });

// Find one
const user = await User.findOne({ email: 'john@example.com' });

// Find by ID
const user = await User.findById('507f1f77bcf86cd799439011');

// Select specific fields
const users = await User.find().select('name email');
const users = await User.find().select('-password');  // Exclude

// Sorting
const users = await User.find().sort({ createdAt: -1 });  // Descending
const users = await User.find().sort('name');  // Ascending

// Pagination
const page = 1;
const limit = 10;
const users = await User.find()
  .skip((page - 1) * limit)
  .limit(limit);

// Count
const count = await User.countDocuments({ isActive: true });
```

### Update

```javascript
// Find and update
const user = await User.findByIdAndUpdate(
  id,
  { name: 'New Name' },
  { new: true, runValidators: true }  // Return updated doc, run validators
);

// Update one
await User.updateOne(
  { email: 'john@example.com' },
  { $set: { name: 'John Updated' } }
);

// Update many
await User.updateMany(
  { isActive: false },
  { $set: { isActive: true } }
);

// Update operators
await User.updateOne({ _id: id }, {
  $set: { name: 'New Name' },      // Set field
  $unset: { tempField: 1 },        // Remove field
  $inc: { loginCount: 1 },         // Increment
  $push: { tags: 'new-tag' },      // Add to array
  $pull: { tags: 'old-tag' },      // Remove from array
  $addToSet: { tags: 'unique' }    // Add if not exists
});
```

### Delete

```javascript
// Find and delete
const user = await User.findByIdAndDelete(id);

// Delete one
await User.deleteOne({ email: 'john@example.com' });

// Delete many
await User.deleteMany({ isActive: false });
```

## 🔍 Query Operators

```javascript
// Comparison
await User.find({ age: { $eq: 25 } });   // Equal
await User.find({ age: { $ne: 25 } });   // Not equal
await User.find({ age: { $gt: 25 } });   // Greater than
await User.find({ age: { $gte: 25 } });  // Greater than or equal
await User.find({ age: { $lt: 25 } });   // Less than
await User.find({ age: { $lte: 25 } });  // Less than or equal
await User.find({ age: { $in: [20, 25, 30] } });  // In array
await User.find({ age: { $nin: [20, 25] } });     // Not in array

// Logical
await User.find({ $and: [{ age: { $gte: 18 } }, { isActive: true }] });
await User.find({ $or: [{ role: 'admin' }, { role: 'moderator' }] });
await User.find({ age: { $not: { $lt: 18 } } });

// Element
await User.find({ phone: { $exists: true } });  // Field exists
await User.find({ age: { $type: 'number' } });  // Field type

// Array
await User.find({ tags: 'javascript' });  // Contains value
await User.find({ tags: { $all: ['javascript', 'nodejs'] } });  // Contains all
await User.find({ tags: { $size: 3 } });  // Array length

// Text search (requires text index)
await User.find({ $text: { $search: 'javascript developer' } });

// Regex
await User.find({ name: { $regex: /john/i } });
await User.find({ email: { $regex: '@gmail.com$' } });
```

## 🔗 Relationships & Population

### Reference (Normalized)

```javascript
// Post schema with author reference
const postSchema = new mongoose.Schema({
  title: String,
  content: String,
  author: {
    type: mongoose.Schema.Types.ObjectId,
    ref: 'User',
    required: true
  },
  comments: [{
    type: mongoose.Schema.Types.ObjectId,
    ref: 'Comment'
  }]
});

// Populate references
const post = await Post.findById(id)
  .populate('author', 'name email')  // Only name and email
  .populate({
    path: 'comments',
    populate: { path: 'author', select: 'name' }  // Nested populate
  });
```

### Embedded (Denormalized)

```javascript
const postSchema = new mongoose.Schema({
  title: String,
  content: String,
  // Embedded comments
  comments: [{
    text: String,
    author: String,
    createdAt: { type: Date, default: Date.now }
  }]
});
```

## ⚡ Middleware (Hooks)

```javascript
// Pre-save hook
userSchema.pre('save', async function(next) {
  // Only hash password if modified
  if (!this.isModified('password')) return next();
  
  this.password = await bcrypt.hash(this.password, 12);
  next();
});

// Post-save hook
userSchema.post('save', function(doc) {
  console.log('User saved:', doc._id);
});

// Pre-find hook
userSchema.pre(/^find/, function(next) {
  // Exclude inactive users by default
  this.find({ isActive: { $ne: false } });
  next();
});

// Pre-remove hook
userSchema.pre('remove', async function(next) {
  // Delete related posts
  await Post.deleteMany({ author: this._id });
  next();
});
```

```javascript
  return await bcrypt.compare(candidatePassword, this.password);
};

// Usage
const user = await User.findOne({ email }).select('+password');
const isMatch = await user.comparePassword(password);

// Static method (on model)
userSchema.statics.findByEmail = function(email) {
  return this.findOne({ email });
};

// Usage
const user = await User.findByEmail('john@example.com');

// Virtual (computed property)
userSchema.virtual('fullName').get(function() {
  return `${this.firstName} ${this.lastName}`;
});

// Enable virtuals in JSON
userSchema.set('toJSON', { virtuals: true });
```

## 🎯 Summary

- MongoDB stores **documents** in **collections**
- **Schemas** define document structure
- **Models** provide CRUD methods
- Use **query operators** for complex queries
- **Populate** for joining related documents
- **Middleware** for pre/post operation logic
- **Methods** for reusable document operations
